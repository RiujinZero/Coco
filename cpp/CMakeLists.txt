cmake_minimum_required(VERSION 3.16)
project(laminar CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

option(LAMINAR_PGO_GEN "Build with -fprofile-generate" OFF)
option(LAMINAR_PGO_USE "Build with -fprofile-use" OFF)

add_library(laminar_core STATIC src/organism.cpp)
target_include_directories(laminar_core PUBLIC include)

if(MSVC)
  target_compile_options(laminar_core PUBLIC /O2 /arch:AVX2 /GL /W3 /EHsc /DNDEBUG /D_CRT_SECURE_NO_WARNINGS)
  target_link_options(laminar_core PUBLIC /LTCG)
else()
  # runbook §1: -O3 -march=native -flto -DNDEBUG; PGO after correctness is frozen
  target_compile_options(laminar_core PUBLIC -O3 -march=native -flto -DNDEBUG -Wall -Wextra -Wno-unused-parameter)
  target_link_options(laminar_core PUBLIC -flto)
  if(LAMINAR_PGO_GEN)
    target_compile_options(laminar_core PUBLIC -fprofile-generate)
    target_link_options(laminar_core PUBLIC -fprofile-generate)
  elseif(LAMINAR_PGO_USE)
    target_compile_options(laminar_core PUBLIC -fprofile-use -fprofile-correction)
    target_link_options(laminar_core PUBLIC -fprofile-use)
  endif()
endif()

foreach(app laminar_synth laminar_bytes laminar_bench laminar_needle laminar_generate laminar_chat laminar_train)
  add_executable(${app} apps/${app}.cpp)
  target_link_libraries(${app} laminar_core)
endforeach()
